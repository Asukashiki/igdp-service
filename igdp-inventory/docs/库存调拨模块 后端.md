# 七、核心业务逻辑

## 7.1 调拨单状态流转

```
DRAFT(草稿) -> SUBMITTED(待审批) -> APPROVED(已完成)
                                  -> REJECTED(未通过)
```

## 7.2 审批通过业务逻辑

当审批通过时，系统需要执行以下操作：

1. 更新调拨单状态为 APPROVED
2. 创建出库单（调出仓库）
   - 出库类型为 TRANSFER（调拨出库）
   - 关联调拨单ID
   - 状态设为 APPROVED
3. 创建入库单（调入仓库）
   - 入库类型为 TRANSFER（调拨入库）
   - 关联调拨单ID
   - 状态设为 APPROVED
4. 执行库存扣减（调出仓库）
5. 执行库存增加（调入仓库）
6. 记录库存流水

## 7.3 核心服务设计

```java
@Service
@Transactional(rollbackFor = Exception.class)
public class TransferServiceImpl implements ITransferService {

    @Autowired
    private TransferMapper transferMapper;
    
    @Autowired
    private TransferDetailMapper detailMapper;
    
    @Autowired
    private IInboundService inboundService;
    
    @Autowired
    private IOutboundService outboundService;
    
    @Autowired
    private IInventoryCoreService inventoryCoreService;

    @Override
    public List<TransferVO> selectTransferList(TransferDTO dto) {
        return transferMapper.selectTransferList(dto);
    }

    @Override
    public TransferVO selectTransferById(Long id) {
        TransferVO vo = transferMapper.selectTransferById(id);
        if (vo != null) {
            vo.setDetailList(detailMapper.selectByTransferId(id));
        }
        return vo;
    }

    @Override
    public Long insertTransfer(TransferDTO dto) {
        // Generate transfer number
        String transferNo = generateTransferNo();
        dto.setTransferNo(transferNo);
        dto.setStatus("DRAFT");
        
        // Insert main record
        InventoryTransfer transfer = BeanUtil.copyProperties(dto, InventoryTransfer.class);
        transferMapper.insert(transfer);
        
        // Insert details
        if (dto.getDetailList() != null) {
            for (InventoryTransferDetail detail : dto.getDetailList()) {
                detail.setTransferId(transfer.getId());
                detailMapper.insert(detail);
            }
        }
        
        return transfer.getId();
    }

    @Override
    public int submitTransfer(Long id) {
        InventoryTransfer transfer = transferMapper.selectById(id);
        if (!"DRAFT".equals(transfer.getStatus())) {
            throw new ServiceException("Only draft orders can be submitted");
        }
        transfer.setStatus("SUBMITTED");
        return transferMapper.updateById(transfer);
    }

    @Override
    public int auditTransfer(Long id, Boolean approved, String auditComment) {
        InventoryTransfer transfer = transferMapper.selectById(id);
        if (!"SUBMITTED".equals(transfer.getStatus())) {
            throw new ServiceException("Only submitted orders can be audited");
        }
        
        // Get current user
        String currentUser = SecurityUtils.getLoginUser().getUsername();
        
        transfer.setAuditBy(currentUser);
        transfer.setAuditTime(new Date());
        transfer.setAuditComment(auditComment);
        
        if (approved) {
            transfer.setStatus("APPROVED");
            transfer.setOutTime(new Date());
            transfer.setInTime(new Date());
            
            // Execute transfer logic
            executeTransfer(transfer);
        } else {
            transfer.setStatus("REJECTED");
        }
        
        return transferMapper.updateById(transfer);
    }

    private void executeTransfer(InventoryTransfer transfer) {
        // Get transfer details
        List<InventoryTransferDetail> details = detailMapper.selectByTransferId(transfer.getId());
        
        // Create outbound order
        InventoryOutbound outbound = new InventoryOutbound();
        outbound.setOutboundNo(generateOutboundNo());
        outbound.setType("TRANSFER");
        outbound.setWarehouseId(transfer.getOutWarehouseId());
        outbound.setWarehouseName(transfer.getOutWarehouseName());
        outbound.setBizNo(transfer.getTransferNo());
        outbound.setStatus("APPROVED");
        outbound.setOperator(transfer.getApplicant());
        outbound.setOrderDate(new Date());
        outboundService.insertOutboundWithDetails(outbound, convertToOutboundDetails(details));
        
        // Create inbound order
        InventoryInbound inbound = new InventoryInbound();
        inbound.setInboundNo(generateInboundNo());
        inbound.setType("TRANSFER");
        inbound.setWarehouseId(transfer.getInWarehouseId());
        inbound.setWarehouseName(transfer.getInWarehouseName());
        inbound.setBizNo(transfer.getTransferNo());
        inbound.setStatus("APPROVED");
        inbound.setOperator(transfer.getApplicant());
        inbound.setOrderDate(new Date());
        inboundService.insertInboundWithDetails(inbound, convertToInboundDetails(details));
        
        // Update related IDs
        transfer.setRelatedOutboundId(outbound.getId());
        transfer.setRelatedInboundId(inbound.getId());
        
        // Execute stock operations
        for (InventoryTransferDetail detail : details) {
            // Deduct from out warehouse
            inventoryCoreService.deductStock(
                transfer.getOutWarehouseId(),
                detail.getProductId(),
                detail.getBatchNo(),
                detail.getRealQty()
            );
            
            // Add to in warehouse
            inventoryCoreService.increaseStock(
                transfer.getInWarehouseId(),
                detail.getProductId(),
                detail.getBatchNo(),
                detail.getRealQty()
            );
        }
    }

    private String generateTransferNo() {
        SimpleDateFormat sdf = new SimpleDateFormat("yyyyMMdd");
        String dateStr = sdf.format(new Date());
        String random = String.format("%04d", new Random().nextInt(10000));
        return "TF" + dateStr + random;
    }
}
```

@RestController
@RequestMapping("/inventory/transfer")
public class TransferController extends BaseController {

    @Autowired
    private ITransferService transferService;

    @PreAuthorize("@ss.hasPermi('inventory:transfer:list')")
    @GetMapping("/list")
    public TableDataInfo list(TransferDTO dto) {
        startPage();
        List<TransferVO> list = transferService.selectTransferList(dto);
        return getDataTable(list);
    }

    @PreAuthorize("@ss.hasPermi('inventory:transfer:view')")
    @GetMapping("/{id}")
    public AjaxResult getInfo(@PathVariable Long id) {
        return AjaxResult.success(transferService.selectTransferById(id));
    }

    @PreAuthorize("@ss.hasPermi('inventory:transfer:add')")
    @PostMapping("/create")
    public AjaxResult add(@RequestBody TransferDTO dto) {
        Long id = transferService.insertTransfer(dto);
        return AjaxResult.success(id);
    }

    @PreAuthorize("@ss.hasPermi('inventory:transfer:edit')")
    @PutMapping
    public AjaxResult edit(@RequestBody TransferDTO dto) {
        return toAjax(transferService.updateTransfer(dto));
    }

    @PreAuthorize("@ss.hasPermi('inventory:transfer:delete')")
    @DeleteMapping("/{ids}")
    public AjaxResult remove(@PathVariable Long[] ids) {
        return toAjax(transferService.deleteTransferByIds(ids));
    }

    @PreAuthorize("@ss.hasPermi('inventory:transfer:submit')")
    @PostMapping("/submit")
    public AjaxResult submit(@RequestBody Map<String, Long> params) {
        Long id = params.get("id");
        return toAjax(transferService.submitTransfer(id));
    }

    @PreAuthorize("@ss.hasPermi('inventory:transfer:audit')")
    @PostMapping("/audit")
    public AjaxResult audit(@RequestBody Map<String, Object> params) {
        Long id = Long.parseLong(params.get("id").toString());
        Boolean approved = Boolean.parseBoolean(params.get("approved").toString());
        String auditComment = params.get("auditComment") != null ? params.get("auditComment").toString() : "";
        return toAjax(transferService.auditTransfer(id, approved, auditComment));
    }
}
